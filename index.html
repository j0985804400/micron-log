const SHEET_NAME = "美光紀錄"; const DRIVE_FOLDER_ID = "1k76WwFQM_aJ5fCNgdhMAZEaUKEgDEg8z"; function doGet() { const sheet = getOrCreateSheet_(); const rows = sheet.getDataRange().getValues(); const records = []; for (let i = 1; i < rows.length; i++) { const rawJson = rows[i][1]; if (!rawJson) continue; try { records.push(JSON.parse(rawJson)); } catch (_) {} } return jsonOut_({ ok: true, records: records }); } function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const body = JSON.parse((e && e.postData && e.postData.contents) || "{}"); const incoming = Array.isArray(body.records) ? body.records : []; const normalized = incoming.map(function (rec, idx) { const copy = Object.assign({}, rec); const rawPhotos = Array.isArray(copy.photos) ? copy.photos : []; copy.photos = rawPhotos.map(function (p, pIdx) { if (typeof p === "string" && p.indexOf("data:image/") === 0) { return saveBase64ToDrive_(p, copy.id || ("R" + Date.now()), idx + 1, pIdx + 1); } return p; }).filter(function (u) { return typeof u === "string" && u.length > 0; }); return copy; }); cleanupUnusedDrivePhotos_(normalized); writeRecordsToSheet_(normalized); return jsonOut_({ ok: true, records: normalized }); } catch (err) { return jsonOut_({ ok: false, error: err && err.message ? err.message : String(err) }); } finally { lock.releaseLock(); } } function saveBase64ToDrive_(dataUrl, recId, recSeq, photoSeq) { const parts = dataUrl.split(","); if (parts.length < 2) return ""; const mimeMatch = parts[0].match(/data:(image\/[a-zA-Z0-9.+-]+);base64/); const mimeType = mimeMatch ? mimeMatch[1] : "image/jpeg"; const ext = mimeType.indexOf("png") >= 0 ? "png" : "jpg"; const bytes = Utilities.base64Decode(parts[1]); const tz = Session.getScriptTimeZone() || "Asia/Taipei"; const stamp = Utilities.formatDate(new Date(), tz, "yyyyMMdd_HHmmss"); const safeId = String(recId || "REC").replace(/[^a-zA-Z0-9_-]/g, ""); const fileName = `${safeId}_${stamp}_${recSeq}_${photoSeq}.${ext}`; const blob = Utilities.newBlob(bytes, mimeType, fileName); const folder = getTargetFolder_(); const file = folder.createFile(blob); try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (_) {} return `https://drive.google.com/file/d/${file.getId()}/view?usp=drivesdk`; } function cleanupUnusedDrivePhotos_(activeRecords) { const activeIds = {}; (activeRecords || []).forEach(function (rec) { const photos = Array.isArray(rec && rec.photos) ? rec.photos : []; photos.forEach(function (url) { const id = extractDriveFileId_(url); if (id) activeIds[id] = true; }); }); const folder = getTargetFolder_(); const files = folder.getFiles(); while (files.hasNext()) { const file = files.next(); if (file.isTrashed()) continue; const fileId = file.getId(); if (!activeIds[fileId]) { file.setTrashed(true); } } } function extractDriveFileId_(url) { if (typeof url !== "string" || !url) return ""; const m1 = url.match(/[?&]id=([a-zA-Z0-9_-]+)/); if (m1) return m1[1]; const m2 = url.match(/\/d\/([a-zA-Z0-9_-]+)/); return m2 ? m2[1] : ""; } function getTargetFolder_() { if (DRIVE_FOLDER_ID && DRIVE_FOLDER_ID.trim()) { try { return DriveApp.getFolderById(DRIVE_FOLDER_ID.trim()); } catch (_) {} } return DriveApp.getRootFolder(); } function getOrCreateSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } if (sheet.getLastRow() === 0) { sheet.appendRow(["更新時間", "紀錄JSON", "NRS單號", "放行單號", "廠區", "工件名稱", "急件", "類別", "備註", "照片連結"]); } return sheet; } function writeRecordsToSheet_(records) { const sheet = getOrCreateSheet_(); const tz = Session.getScriptTimeZone() || "Asia/Taipei"; const nowStr = Utilities.formatDate(new Date(), tz, "yyyy/MM/dd HH:mm:ss"); const rows = records.map(function (r) { return [ nowStr, JSON.stringify(r), r.nrs || "", r.passNo || "", r.plant || "", r.part || "", r.urgent ? "急件" : "", r.cat || "", r.note || "", Array.isArray(r.photos) ? r.photos.join("\n") : "" ]; }); const lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).clearContent(); } if (rows.length > 0) { sheet.getRange(2, 1, rows.length, rows[0].length).setValues(rows); } } function jsonOut_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
